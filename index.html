<script>

    // 常見分潤參數關鍵字 (不區分大小寫)

    const AFFILIATE_KEYWORDS = [

        'affiliate', 'aff_id', 'affid', 'affiliate_id', 'aff_sub',

        'ref', 'referral', 'referrer', 'refer',

        'partner', 'partner_id', 'pub_id', 'publisher',

        'click_id', 'clickid', 'c_id',

        'utm_source',   // 若包含 'aff', 'partner', 'ref' 等值需再細判

        'source=aff', 'medium=affiliate',

        'tag',         // 常用在 Amazon

        'sr',          // 某些聯盟

        'afp', 'afclick', 'afftrack', 'campaign',

        'sid', 'subid', 'sub_id'

    ];

    // 進階: 檢查 utm_source 的值是否疑似分潤

    function isUtmSourceAffiliate(value) {

        if (!value) return false;

        const lowerVal = value.toLowerCase();

        const affPatterns = ['aff', 'partner', 'ref', 'referral', 'shareasale', 'cj', 'impact', 'rakuten'];

        return affPatterns.some(p => lowerVal.includes(p));

    }

    // 解析最終網址參數

    function parseFinalUrl(finalUrl) {

        try {

            const urlObj = new URL(finalUrl);

            const params = urlObj.searchParams;

            const foundParams = [];

            // 檢查每個參數名稱與值

            for (let [key, value] of params.entries()) {

                const lowerKey = key.toLowerCase();

                // 參數名稱直接命中關鍵字

                if (AFFILIATE_KEYWORDS.some(kw => lowerKey.includes(kw))) {

                    foundParams.push({ key, value, reason: `參數名稱包含 "${key}"` });

                }

                // 特別處理 utm_source

                else if (lowerKey === 'utm_source' && isUtmSourceAffiliate(value)) {

                    foundParams.push({ key, value, reason: `utm_source 值疑似分潤: "${value}"` });

                }

                // 檢查參數值是否像分潤ID (單純輔助, 避免誤判)

                else if (value && (value.match(/affid|ref=|partner=/i) || /^[0-9]{5,}$/.test(value) && (lowerKey.includes('id') || lowerKey === 'pid'))) {

                    // 保守一點，只當 key 也含 id 時才列為可疑

                    if (lowerKey.includes('id') || lowerKey === 'pid' || lowerKey === 'sid') {

                        foundParams.push({ key, value, reason: `參數值疑似聯盟ID` });

                    }

                }

            }

            // 若沒有任何發現，但網址中有常見聯盟網域特徵 (簡易)

            let domainHint = false;

            if (urlObj.hostname.includes('shareasale') || urlObj.hostname.includes('avantlink') ||

                urlObj.hostname.includes('impactradius') || urlObj.hostname.includes('rakuten')) {

                domainHint = true;

            }

            return { foundParams, domainHint, fullUrl: finalUrl, hostname: urlObj.hostname };

        } catch(e) {

            return { foundParams: [], domainHint: false, fullUrl: finalUrl, error: 'URL解析失敗' };

        }

    }

    // 透過代理獲取最終重定向網址

    async function getFinalUrl(shortUrl) {

        // 注意：由於 CORS 限制，我們無法直接用 fetch 取得重定向目標。

        // 解決方法：使用一個支援 CORS 的無狀態重定向追蹤 API

        // 範例使用 "https://api.allorigins.win/raw?url=" 但那個無法跟隨重定向。

        // 改用 cors-anywhere 搭配 mode: 'no-cors' 也無法讀取 Location。

        // 因此這裡改用 「伺服器端解析」不可能純前端做到? 其實可以用一個公開的 unshorten.me 類服務

        // 最簡單穩定: 使用「https://unshorten.me/json/」免費 API (不需 key)

       

        // 嘗試使用 unshorten.me API

        const apiUrl = `https://unshorten.me/json/${encodeURIComponent(shortUrl)}`;

        try {

            const response = await fetch(apiUrl);

            if (!response.ok) throw new Error('API 請求失敗');

            const data = await response.json();

            if (data && data.resolved_url) {

                return data.resolved_url;

            } else {

                throw new Error('無法解析最終網址');

            }

        } catch (err) {

            console.error(err);

            throw new Error(`解析失敗: ${err.message}. 請手動檢查最終網址參數。`);

        }

    }

    // 顯示結果

    function displayResult(finalUrl, analysis) {

        const resultDiv = document.getElementById('resultArea');

        resultDiv.style.display = 'block';

        const isAffiliate = analysis.foundParams.length > 0 || analysis.domainHint;

        const badgeClass = isAffiliate ? 'affiliate' : 'safe';

        const badgeText = isAffiliate ? '⚠️ 偵測到分潤跡象' : '✅ 無明顯分潤特徵';

        let paramsHtml = '';

        if (analysis.foundParams.length > 0) {

            paramsHtml = `<div style="margin-top: 12px;"><strong>🔎 可疑參數：</strong>

                            <ul style="margin-top: 8px;">` +

                            analysis.foundParams.map(p => `<li><code>${p.key}=${p.value}</code><br> <span style="font-size:0.75rem;">➜ ${p.reason}</span></li>`).join('') +

                         `</ul></div>`;

        } else {

            paramsHtml = `<div style="margin-top: 12px;">✔️ 未發現典型分潤參數。</div>`;

        }

        if (analysis.domainHint && analysis.foundParams.length === 0) {

            paramsHtml += `<div>🏷️ 網域屬於聯盟平台 (${analysis.hostname})，可能帶有隱藏分潤。</div>`;

        }

        resultDiv.innerHTML = `

            <div class="badge ${badgeClass}">${badgeText}</div>

            <div><strong>🔗 最終網址：</strong><br><span style="font-size:0.8rem;">${escapeHtml(analysis.fullUrl)}</span></div>

            <hr>

            ${paramsHtml}

            <div style="margin-top: 12px; font-size:0.8rem; color:#475569;">

            📌 分潤常見參數：affiliate, ref, partner, click_id, utm_source=aff, sid, subid 等

            </div>

        `;

    }

    function escapeHtml(str) {

        return str.replace(/[&<>]/g, function(m) {

            if (m === '&') return '&amp;';

            if (m === '<') return '&lt;';

            if (m === '>') return '&gt;';

            return m;

        });

    }

    async function handleCheck() {

        const input = document.getElementById('shortUrl').value.trim();

        if (!input) {

            alert('請輸入短網址');

            return;

        }

        // 基本 URL 格式檢查

        let urlToCheck = input;

        if (!urlToCheck.startsWith('http://') && !urlToCheck.startsWith('https://')) {

            urlToCheck = 'https://' + urlToCheck;

        }

        const resultDiv = document.getElementById('resultArea');

        resultDiv.style.display = 'block';

        resultDiv.innerHTML = `<div>⏳ 解析中，請稍候 (呼叫 unshorten.me API)...</div>`;

        try {

            const finalUrl = await getFinalUrl(urlToCheck);

            const analysis = parseFinalUrl(finalUrl);

            displayResult(finalUrl, analysis);

        } catch (err) {

            resultDiv.innerHTML = `<div class="badge error">❌ 偵測錯誤</div>

                                   <div>${escapeHtml(err.message)}</div>

                                   <div style="margin-top:12px;">👉 替代方法：請手動點擊短網址，然後查看瀏覽器網址列中的參數。<br>

                                    例如有 <code>?aff_id=xxx</code> 或 <code>&partner=</code> 即為分潤。</div>`;

        }

    }

    document.getElementById('checkBtn').addEventListener('click', handleCheck);
